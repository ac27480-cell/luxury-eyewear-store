function toggleStores(brand, containerId) { const container = document.getElementById(containerId); if (!container) return; if (!container.classList.contains('hidden')) { container.classList.add('hidden'); return; } const matchingStores = STORE_DATABASE.filter(store => store.brands.includes(brand)); let html = '
Online E-Commerce Shops
'; matchingStores.filter(s => s.type === 'online').forEach(store => { html += `

${store.name}

${store.desc}

${store.status} `; }); html += '
Optiker Fachgeschäfte & Partner
'; matchingStores.filter(s => s.type === 'store').forEach(store => { html += `

${store.name} ${store.state}

${store.desc}

${store.status} `; }); container.innerHTML = html; container.classList.remove('hidden'); }
